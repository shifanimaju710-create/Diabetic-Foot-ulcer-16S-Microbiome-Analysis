  arrange(desc(Total)) %>%
  mutate(
    Taxon = Genus,
    id = row_number(),
    angle = 2 * pi * (id - 1) / n(),
    Group = rep(
      c("Group 1", "Group 2", "Group 3", "Group 4"),
      length.out = n()
    )
  )
