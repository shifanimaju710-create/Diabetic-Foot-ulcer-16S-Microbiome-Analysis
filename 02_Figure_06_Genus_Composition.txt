tax$Genus <- stringr::str_extract(
  tax$`#OTU ID`,
  "g__[^;]+"
)
tax$Genus <- sub("^g__", "", tax$Genus)
sort(unique(tax$Genus))
genus_long <- tax %>%
  dplyr::select(
    Genus,
    DF_01,
    DF_02,
    DF_03,
    DF_04,
    DF_05
  ) %>%
  tidyr::pivot_longer(
    cols = starts_with("DF_"),
    names_to = "Sample",
    values_to = "Abundance"
  )
genus_long$Abundance <- as.numeric(genus_long$Abundance)
genus_long$Genus[
  is.na(genus_long$Genus) |
    genus_long$Genus == ""
] <- "Unclassified"
genus_sum <- genus_long %>%
  group_by(Sample, Genus) %>%
  summarise(
    Abundance = sum(Abundance, na.rm = TRUE),
    .groups = "drop"
  )
genus_sum <- genus_sum %>%
  group_by(Sample) %>%
  mutate(
    Relative = 100 * Abundance / sum(Abundance)
  ) %>%
  ungroup()
top10_genus <- genus_sum %>%
  filter(Genus != "Unclassified") %>%
  group_by(Genus) %>%
  summarise(
    Total = sum(Abundance),
    .groups = "drop"
  ) %>%
  arrange(desc(Total)) %>%
  slice_head(n = 10) %>%
  pull(Genus)
top10_genus
genus_plot <- genus_sum %>%
  mutate(
    Genus_plot = ifelse(
      Genus %in% top10_genus,
      Genus,
      "Other"
    )
  ) %>%
  group_by(Sample, Genus_plot) %>%
  summarise(
    Relative = sum(Relative, na.rm = TRUE),
    .groups = "drop"
  )
genus_plot$Sample <- factor(
  genus_plot$Sample,
  levels = c(
    "DF_01",
    "DF_02",
    "DF_03",
    "DF_04",
    "DF_05"
  )
)
genus_names <- unique(genus_plot$Genus_plot)
genus_palette <- c(
  "#2B6CA3",
  "#43AAA4",
  "#F6D55C",
  "#F05A40",
  "#8E44AD",
  "#5DA5DA",
  "#58CF87",
  "#F5B041",
  "#A569BD",
  "#566573",
  "#D35400"
)
genus_cols <- setNames(
  genus_palette[seq_along(genus_names)],
  genus_names
)
pGenusA <- ggplot(
  genus_plot,
  aes(
    x = Sample,
    y = Relative,
    fill = Genus_plot
  )
)
pGenusA <- pGenusA + geom_col(
  width = 0.78,
  colour = "white",
  linewidth = 0.2
)
pGenusA <- pGenusA + scale_fill_manual(
  values = genus_cols,
  name = NULL
)
pGenusA <- pGenusA + scale_y_continuous(
  limits = c(0, 100),
  breaks = seq(0, 100, 20),
  labels = function(x) paste0(x, "%"),
  expand = c(0, 0)
)
pGenusA <- pGenusA + labs(
  title = "Top genera across DFU samples",
  x = NULL,
  y = "Relative abundance (%)"
)
pGenusA <- pGenusA + theme_classic(base_size = 9) +
