# ------------------------------------------------------------
# 2. IMPORT DATA
# ------------------------------------------------------------

# Select raw_microbiome.xlsx(1).xlsx
file_path <- file.choose()

micro <- readxl::read_excel(file_path)

# Check columns
print(colnames(micro))


# ------------------------------------------------------------
# 3. EXTRACT TAXONOMIC RANKS
# ------------------------------------------------------------

tax_tree <- micro

tax_tree$Phylum <- stringr::str_extract(
  tax_tree$`#OTU ID`,
  "p__[^;]+"
)

tax_tree$Class <- stringr::str_extract(
  tax_tree$`#OTU ID`,
  "c__[^;]+"
)

tax_tree$Order <- stringr::str_extract(
  tax_tree$`#OTU ID`,
  "o__[^;]+"
)

tax_tree$Family <- stringr::str_extract(
  tax_tree$`#OTU ID`,
  "f__[^;]+"
)

tax_tree$Genus <- stringr::str_extract(
  tax_tree$`#OTU ID`,
  "g__[^;]+"
)


# Remove prefixes

tax_tree$Phylum <- sub("^p__", "", tax_tree$Phylum)
tax_tree$Class  <- sub("^c__", "", tax_tree$Class)
tax_tree$Order  <- sub("^o__", "", tax_tree$Order)
tax_tree$Family <- sub("^f__", "", tax_tree$Family)
tax_tree$Genus  <- sub("^g__", "", tax_tree$Genus)


# ------------------------------------------------------------
# 4. PREPARE TAXONOMIC HIERARCHY
# ------------------------------------------------------------

tree_tax <- tax_tree %>%
  dplyr::select(
    Phylum,
    Class,
    Order,
    Family,
    Genus
  ) %>%
  dplyr::filter(
    !is.na(Phylum),
    !is.na(Genus),
    Genus != ""
  ) %>%
  dplyr::distinct()


# Fill missing intermediate ranks

tree_tax <- tree_tax %>%
  dplyr::mutate(
    
    Class = ifelse(
      is.na(Class) | Class == "",
      "Unclassified_class",
      Class
    ),
    
    Order = ifelse(
      is.na(Order) | Order == "",
      "Unclassified_order",
      Order
    ),
    
    Family = ifelse(
      is.na(Family) | Family == "",
      "Unclassified_family",
      Family
    )
  )


# ------------------------------------------------------------

# 8. CONVERT TAXONOMIC HIERARCHY TO NEWICK TREE
# ------------------------------------------------------------

children <- split(
  tree_edges$to,
  tree_edges$from
)

tip_lookup <- setNames(
  tip_map$tip_id,
  tip_map$G
)


make_newick <- function(node) {
  
  kids <- children[[node]]
  
  if (is.null(kids) || length(kids) == 0) {
    
    id <- unname(
      tip_lookup[node]
    )
    
    if (
      length(id) == 0 ||
      is.na(id)
    ) {
      
      stop(
        paste(
          "Terminal node not found:",
          node
        )
      )
    }
    
    return(id)
  }
  
  child_text <- vapply(
    kids,
    make_newick,
    character(1)
  )
  
  paste0(
    "(",
    paste(
      child_text,
      collapse = ","
    ),
    ")"
  )
}

  ";"
